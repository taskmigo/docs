import { Accordion, Accordions } from 'fumadocs-ui/components/accordion';

<Accordions type='single'>
  <Accordion title='What identifies a resource?' id='resource-identity'>
    `(kind, metadata.name)`. See [Resource model](/versions/v0/resources#identity-and-revisions).
  </Accordion>
  <Accordion title='How do I pin an imported resource version?' id='import-version'>
    Append a tag constraint, such as `component/header@1.2.0` or `component/header@^1.2.0`. See
    [References](/versions/v0/resources#references).
  </Accordion>
  <Accordion title='How do I reuse one Component across Pages?' id='reuse-component'>
    Import it from each Page and pass Page-specific `props`. See
    [Component](/versions/v0/manifests/v0-component#use-the-component-in-a-page).
  </Accordion>
  <Accordion title='Where can I use an Expression?' id='expression-scope'>
    Fields marked with the Expression tag in their Type Table. See [Expressions](/versions/v0/expressions).
  </Accordion>
  <Accordion title='Where is a Page route defined?' id='page-route'>
    In its Application. See [Paths and routes](/versions/v0/manifests/v0-application#paths-and-routes).
  </Accordion>
  <Accordion title='Why did my change not become active?' id='invalid-change'>
    See [Runtime lifecycle](/versions/v0/runtime).
  </Accordion>
  <Accordion title='What happens to a running Workflow after an update?' id='workflow-version-pinning'>
    See [Execution and revision pinning](/versions/v0/manifests/v0-workflow#execution-and-revision-pinning).
  </Accordion>
  <Accordion title='Which contracts can I use now?' id='defined-kinds'>
    See [Product status](/versions/v0/status).
  </Accordion>
</Accordions>
